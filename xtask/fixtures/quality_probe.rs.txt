// The constructs the quality count models, one function each, held to
// clippy's counts in quality_probe.clippy.tsv (scripts/quality-probe-clippy.sh).
#![allow(dead_code, unused)]
pub fn plain(x: i32) -> i32 {
    x + 1
}
pub fn one_if(x: i32) -> i32 {
    if x > 0 { 1 } else { 2 }
}
pub fn one_assert(x: i32) {
    assert!(x > 0);
}
pub fn one_assert_eq(x: i32) {
    assert_eq!(x, 1);
}
pub fn one_debug_assert(x: i32) {
    debug_assert!(x > 0);
}
pub fn one_matches(x: Option<i32>) -> bool {
    matches!(x, Some(_))
}
pub fn one_for(v: &[i32]) -> i32 {
    let mut s = 0;
    for x in v {
        s += x;
    }
    s
}
pub fn one_while(mut x: i32) {
    while x > 0 {
        x -= 1;
    }
}
pub fn one_while_let(mut v: Vec<i32>) {
    while let Some(_) = v.pop() {}
}
pub fn one_loop(mut x: i32) {
    loop {
        x -= 1;
        if x < 0 {
            break;
        }
    }
}
pub fn try_result(x: Result<i32, String>) -> Result<i32, String> {
    let y = x?;
    Ok(y)
}
pub fn try_option(x: Option<i32>) -> Option<i32> {
    let y = x?;
    Some(y)
}
pub fn two_try_option(x: Option<i32>) -> Option<i32> {
    let y = x?;
    let z = x?;
    Some(y + z)
}
pub fn let_else_bool(x: Option<i32>) -> bool {
    let Some(_) = x else {
        return false;
    };
    true
}
pub fn let_else_two_ifs(x: Option<i32>, y: bool, z: bool) -> bool {
    let Some(_) = x else {
        if y {
            return false;
        }
        if z {
            return false;
        }
        return true;
    };
    true
}
pub fn and_or(a: bool, b: bool, c: bool) -> bool {
    a && b || c
}
pub fn one_return(x: i32) -> i32 {
    if x > 0 {
        return 1;
    }
    2
}
pub fn two_returns_bool(x: i32) -> bool {
    if x > 0 {
        return true;
    }
    if x < -5 {
        return false;
    }
    true
}
pub fn match_two(x: Option<i32>) -> i32 {
    match x {
        Some(v) => v,
        None => 0,
    }
}
pub fn match_one(x: (i32, i32)) -> i32 {
    match x {
        (a, b) => a + b,
    }
}
pub fn match_guard(x: Option<i32>) -> i32 {
    match x {
        Some(v) if v > 0 => v,
        Some(_) => 1,
        None => 0,
    }
}
pub fn if_let_chain(x: Option<i32>, y: Option<i32>) -> i32 {
    if let Some(a) = x
        && let Some(b) = y
    {
        a + b
    } else {
        0
    }
}
pub fn outer_with_nested(x: i32) -> i32 {
    fn inner(y: i32) -> i32 {
        if y > 0 { 1 } else { 0 }
    }
    inner(x)
}
pub fn format_try(x: Result<i32, String>) -> Result<String, String> {
    Ok(format!("{}", x?))
}
pub fn if_in_macro_arg(x: i32) -> String {
    format!("{}", if x > 0 { 1 } else { 0 })
}
pub fn outer_closure_two_ifs(v: &[i32]) -> i32 {
    let f = |x: i32| -> i32 { if x > 0 { 1 } else if x < 0 { -1 } else { 0 } };
    f(v.len() as i32)
}
pub fn outer_closure_returns(v: &[i32]) -> i32 {
    let f = |x: i32| -> i32 {
        if x > 0 {
            return 1;
        }
        0
    };
    if v.is_empty() {
        return 0;
    }
    f(1)
}
pub fn dbg_macro(x: i32) -> i32 {
    dbg!(x)
}
pub fn panic_in_else(x: i32) -> i32 {
    if x > 0 { 1 } else { panic!("no") }
}
pub fn comments_and_blanks(x: i32) -> i32 {
    // a comment line

    let y = x + 1; // trailing
    /* a block
    comment */
    y
}
